const dateGroup = document.createElement('div'); dateGroup.className = 'pay-calendar-day__date-group'; const dayName = document.createElement('span'); dayName.className = 'pay-calendar-day__name'; dayName.textContent = String( day.dayName || day.shortDayName || '' ); const date = document.createElement('span'); date.className = 'pay-calendar-day__date'; date.textContent = String( day.dateLabel || '' ); dateGroup.appendChild(dayName); dateGroup.appendChild(date); const totals = document.createElement('div'); totals.className = 'pay-calendar-day__totals'; const pay = document.createElement('strong'); pay.className = 'pay-calendar-day__pay'; pay.textContent = day.hasWork ? this.formatMoney(day.totalPay) : '—'; const hours = document.createElement('span'); hours.className = 'pay-calendar-day__hours'; hours.textContent = day.hasWork ? this.formatHours(day.totalHours) : 'No work'; totals.appendChild(pay); totals.appendChild(hours); header.appendChild(dateGroup); header.appendChild(totals); const entriesContainer = document.createElement('div'); entriesContainer.className = 'pay-calendar-day__entries'; const entries = Array.isArray(day.entries) ? day.entries : []; if (entries.length === 0) { entriesContainer.appendChild( this.createEmptyEntry() ); } else { entries.forEach(function(entry) { entriesContainer.appendChild( PayTrackerPayCalendarService .createShiftCard(entry || {}) ); }); } card.appendChild(header); card.appendChild(entriesContainer); return card; }, createShiftCard: function(entry) { const employerKey = this.normalizeEmployerKey( entry.employerKey || entry.employerName ); const card = document.createElement('div'); card.className = 'pay-calendar-shift'; if (employerKey) { card.classList.add( 'pay-calendar-shift--' + employerKey ); } const header = document.createElement('div'); const pay = document.createElement('strong'); pay.className = 'pay-calendar-shift__pay'; pay.textContent = this.formatMoney(entry.pay); header.appendChild(employer); header.appendChild(pay); const shiftName = document.createElement('p'); const hours = document.createElement('span'); hours.className = 'pay-calendar-shift__detail'; hours.textContent = this.formatHours(entry.hours); const enhancement = document.createElement('span'); enhancement.className = 'pay-calendar-shift__enhancement'; enhancement.textContent = this.formatEnhancement( entry.enhancement, entry.shiftName ); details.appendChild(hours); details.appendChild(enhancement); card.appendChild(header); card.appendChild(shiftName); card.appendChild(details); return card; }, createEmptyEntry: function() { const element = document.createElement('div'); element.className = 'pay-calendar-day__empty-entry'; element.textContent = 'No shifts recorded'; totalShifts += PayTrackerPayCalendarService.toNumber( currentDay.shiftCount ); totalHours += PayTrackerPayCalendarService.toNumber( currentDay.totalHours ); totalPay += PayTrackerPayCalendarService.toNumber( currentDay.totalPay ); if (currentDay.hasWork) { workingDays += 1; } }); this.setText( 'pay-calendar-shift-count', totalShifts === 1 ? '1 shift' : totalShifts + ' shifts' ); this.setText( 'pay-calendar-total-pay', this.formatMoney(totalPay) ); this.setText( 'pay-calendar-total-hours', this.formatHours(totalHours) ); this.setText( 'pay-calendar-working-days', workingDays ); } ); if (populatedDays.length === 0) { return '—'; } populatedDays.sort(function(first, second) { const shiftDifference = PayTrackerPayCalendarService.toNumber( second.shiftCount ) - PayTrackerPayCalendarService.toNumber( first.shiftCount ); return Number.isFinite(number) ? number : 0; }, clearElement: function(element) { while (element.firstChild) { element.removeChild( element.firstChild ); } }, renderMessage: function( container, message ) { const element = document.createElement('div'); element.className = 'pay-calendar-loading'; element.setAttribute( 'role', 'status' ); element.textContent = String(message); container.appendChild(element); }, setText: function(elementId, value) { const element = document.getElementById(elementId); if (!element) { return; } element.textContent = value === null || typeof value === 'undefined' ? '' : String(value); } }); window.PayTrackerPayCalendarService = PayTrackerPayCalendarService;
